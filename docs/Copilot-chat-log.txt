module dasflex.webutil;

import std.stdio;
import std.file : append, exists, mkdirRecurse, isDir;
import std.datetime : Clock, SysTime, DateTime;
import std.process : environment;
import std.string : format, replace;
import std.path : dirName, baseName;

struct DasLog {
    string path;
    string ip;
    this(string path = null, string ip = null) {
        this.path = path;
        this.ip = ip;
        if (path.length) {
            auto pdir = dirName(path);
            if (pdir.length && !isDir(pdir)) mkdirRecurse(pdir);
        }
    }
    void write(string msg) {
        auto ts = Clock.currTime;
        auto line = format("[%04d-%02d-%02d %02d:%02d:%02d] pid=%s ip=%s %s\n",
            ts.year, ts.month, ts.day, ts.hour, ts.minute, ts.second,
            environment.get("SERVER_PID", environment.get("GATEWAY_INTERFACE", "")),
            ip.length ? ip : environment.get("REMOTE_ADDR"),
            msg);
        if (path.length) {
            append(path, line);
        } else {
            // No log path → write to stderr to keep CGI outputs clean
            stderr.write(line);
        }
    }
}

struct WebIO {
    // Lightweight HTTP helpers
    void writeHeaders(string contentType, bool noCache = false) {
        import std.array : appender;
        auto out = appender!string;
        out.put("Status: 200 OK\r\n");
        out.put("Content-Type: " ~ contentType ~ "\r\n");
        if (noCache) {
            out.put("Cache-Control: no-store, no-cache, must-revalidate\r\n");
            out.put("Pragma: no-cache\r\n");
        }
        out.put("\r\n");
        stdout.write(out.data);
    }

    void queryError(ref DasLog log, string msg) {
        stdout.write("Status: 400 Bad Request\r\nContent-Type: text/plain; charset=utf-8\r\n\r\n");
        stdout.write(msg ~ "\n");
        log.write("QUERY ERROR: " ~ msg.replace("\n", " "));
    }

    void notFoundError(ref DasLog log, string msg) {
        stdout.write("Status: 404 Not Found\r\nContent-Type: text/plain; charset=utf-8\r\n\r\n");
        stdout.write(msg ~ "\n");
        log.write("NOT FOUND: " ~ msg.replace("\n", " "));
    }

    void serverError(ref DasLog log, string msg) {
        stdout.write("Status: 500 Internal Server Error\r\nContent-Type: text/plain; charset=utf-8\r\n\r\n");
        stdout.write(msg ~ "\n");
        log.write("SERVER ERROR: " ~ msg.replace("\n", " "));
    }

    string getUrl() {
        auto sn = environment.get("SCRIPT_NAME");
        auto qs = environment.get("QUERY_STRING");
        auto pi = environment.get("PATH_INFO");
        string url = sn ~ (pi.length ? pi : "");
        if (qs.length) url ~= "?" ~ qs;
        return url;
    }

    // Logger factory; mirrors usage in cgimain
    DasLog DasLogFile(string path = null, string ip = null) {
        return DasLog(path, ip);
    }
}

struct Misc {
    void envPathMunge(string key, string extra) {
        if (!extra.length) return;
        auto old = environment.get(key);
        if (old.length) environment[key] = extra ~ ":" ~ old;
        else environment[key] = extra;
    }

    // TODO: replace with stricter checks if needed
    bool checkParams(ref DasLog log, auto form) {
        import std.regex : regex, match;
        auto suspicious = regex(`(?:\b(?:;|&&|\|\|)|[`"\\]|<(?:script|\?|%)|%[0-9A-Fa-f]{2})`);
        foreach (k, v; form.params) {
            if (match(v, suspicious)) {
                log.write(format("Param suspicious: %s=%s", k, v));
                return false;
            }
        }
        return true;
    }
}

struct UType {
    WebIO webio;
    Misc  misc;
}

// Factory used by cgimain.d
UType dasflexGetU() {
    UType u;
    u.webio = WebIO();
    u.misc  = Misc();
    return u;
}

Example handler (dasflex/handlers/resource.d)
This serves files below RESOURCE_PATH for /static/* and other resource paths. It shows how to implement IHandler and self‑register with the dotted name dasflex.handlers.resource.

module dasflex.handlers.resource;

import std.stdio;
import std.file   : exists, read, isFile;
import std.path   : buildPath, relativePath, absolutePath, dirName;
import std.string : startsWith, strip, endsWith, format;
import std.uri    : decodeComponent;
import std.conv   : to;
import std.algorithm : canFind;

import cgimain : IHandler, registerHandler; // interface + registry

final class ResourceHandler : IHandler {
    override int handleReq(ref U, string sReqType,
                           const string[string] dConf,
                           auto ref fLog, // DasLog
                           const typeof(form) form, // FieldStorage
                           string sPathInfo)
    {
        // Only handle paths that resolve under RESOURCE_PATH
        auto root = dConf["RESOURCE_PATH"];
        if (!root.length) {
            U.webio.serverError(fLog, "RESOURCE_PATH not configured");
            return 1;
        }

        // Map PATH_INFO to a file path
        // Examples:
        //   /static/logo.png -> <root>/static/logo.png
        //   /id.txt          -> <root>/id.txt
        auto rel = sPathInfo.length ? sPathInfo[1 .. $] : ""; // trim leading '/'
        // Basic traversal prevention (already checked in cgimain, but defense-in-depth)
        if (rel.canFind("..")) {
            U.webio.queryError(fLog, "Path traversal rejected");
            return 2;
        }

        auto path = buildPath(root, rel);
        if (!isFile(path)) {
            U.webio.notFoundError(fLog, format("Resource not found: %s", rel));
            return 3;
        }

        // Content-Type: guess by extension (minimal)
        string ctype = "application/octet-stream";
        if (rel.endsWith(".html")) ctype = "text/html; charset=utf-8";
        else if (rel.endsWith(".txt")) ctype = "text/plain; charset=utf-8";
        else if (rel.endsWith(".json")) ctype = "application/json";
        else if (rel.endsWith(".png")) ctype = "image/png";
        else if (rel.endsWith(".jpg") || rel.endsWith(".jpeg")) ctype = "image/jpeg";
        else if (rel.endsWith(".csv")) ctype = "text/csv; charset=utf-8";

        U.webio.writeHeaders(ctype);
        auto data = read(path);
        stdout.rawWrite(data);
        return 0;
    }

    override @property string __file__() { return __FILE__; }
    override @property string __name__() { return "dasflex.handlers.resource"; }
}

// Register at load time
static this() {
    registerHandler("dasflex.handlers.resource", new ResourceHandler());
}

module dasflex.webutil.misc;

import std.process : environment;
import std.regex   : regex, match;
import std.string  : format;

struct Misc {
    /// Prepend extra to environment PATH-like keys
    void envPathMunge(string key, string extra) {
        if (!extra.length) return;
        auto old = environment.get(key);
        environment[key] = old.length ? (extra ~ ":" ~ old) : extra;
    }

    /// Very basic injection guard; expand in parity with Python logic
    bool checkParams(ref typeof(this).DasLogProto log, auto form) {
        // NOTE: in D we can’t forward-declare DasLog from webio; use alias trick below.
        auto suspicious = regex(`(?:\b(?:;|&&|\|\|)|[`"\\]|<(?:script|\?|%)|%[0-9A-Fa-f]{2})`);
        foreach (k, v; form.params) {
            if (match(v, suspicious)) {
                log.write(format("Param suspicious: %s=%s", k, v));
                return false;
            }
        }
        return true;
    }

    // Provide a compatible proto for calling .write() on the log
    alias DasLogProto = struct { void write(string) {} };
}

module dasflex.webutil.mime;

import std.string : endsWith;

/// Return a MIME content-type from a (lowercased) filename or extension.
string guessContentType(string name) @safe {
    if (name.endsWith(".html")) return "text/html; charset=utf-8";
    if (name.endsWith(".txt"))  return "text/plain; charset=utf-8";
    if (name.endsWith(".json")) return "application/json";
    if (name.endsWith(".csv"))  return "text/csv; charset=utf-8";
    if (name.endsWith(".xml"))  return "application/xml";
    if (name.endsWith(".png"))  return "image/png";
    if (name.endsWith(".jpg") || name.endsWith(".jpeg")) return "image/jpeg";
    if (name.endsWith(".gif"))  return "image/gif";
    if (name.endsWith(".svg"))  return "image/svg+xml";
    if (name.endsWith(".css"))  return "text/css; charset=utf-8";
    if (name.endsWith(".js"))   return "application/javascript";
    return "application/octet-stream";
}

final class IntroHandler : IHandler {
    override int handleReq(ref UType U, string sReqType,
                           const string[string] dConf,
                           ref DasLog fLog,
                           const typeof(form) form, // FieldStorage
                           string sPathInfo)
    {
        auto site = dConf.get("SERVER_NAME", "DASFlex Server");
        auto id   = dConf.get("SERVER_ID",   "unknown");

        U.webio.writeHeaders("text/plain; charset=utf-8", true);
        stdout.write(format(
            "Welcome to %s (%s)\n\nVisit /source/, /catalog.json, /id.txt, or try ?server=list\n",
            site, id));

        fLog.write("Intro page served");
        return 0;
    }

    override @property string __file__() { return __FILE__; }
    override @property string __name__() { return "dasflex.handlers.intro"; }
